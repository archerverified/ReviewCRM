```typescript
import { ButtonHTMLAttributes, ReactNode, forwardRef } from 'react';
import { cn } from '@/lib/utils';

interface ButtonProps extends ButtonHTMLAttributes<HTMLButtonElement> {
  variant?: 'primary' | 'secondary' | 'ghost' | 'danger';
  size?: 'sm' | 'md' | 'lg';
  loading?: boolean;
  leftIcon?: ReactNode;
  rightIcon?: ReactNode;
  children: ReactNode;
}

export const Button = forwardRef<HTMLButtonElement, ButtonProps>(
  ({ variant = 'primary', size = 'md', loading, leftIcon, rightIcon, disabled, children, className, ...props }, ref) => {
    const baseStyles = 'inline-flex items-center justify-center font-medium transition-colors focus:outline-none focus:ring-2 focus:ring-blue-500 focus:ring-offset-2 disabled:opacity-50 disabled:cursor-not-allowed';
    
    const variantStyles = {
      primary: 'bg-black text-white hover:bg-gray-800',
      secondary: 'border-2 border-black/10 bg-white text-gray-900 hover:bg-black/5',
      ghost: 'text-blue-600 hover:bg-blue-50',
      danger: 'bg-red-600 text-white hover:bg-red-700'
    };
    
    const sizeStyles = {
      sm: 'text-sm px-3 py-1.5 rounded-lg',
      md: 'px-4 py-2 rounded-xl',
      lg: 'px-6 py-3 text-lg rounded-xl'
    };
    
    return (
      <button
        ref={ref}
        className={cn(baseStyles, variantStyles[variant], sizeStyles[size], className)}
        disabled={disabled || loading}
        {...props}
      >
        {loading && (
          <svg className="animate-spin -ml-1 mr-2 h-4 w-4" fill="none" viewBox="0 0 24 24">
            <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" />
            <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z" />
          </svg>
        )}
        {!loading && leftIcon && <span className="mr-2">{leftIcon}</span>}
        {children}
        {rightIcon && <span className="ml-2">{rightIcon}</span>}
      </button>
    );
  }
);

Button.displayName = 'Button';
```

```typescript
'use client';

import { ReactNode, useEffect } from 'react';
import { createPortal } from 'react-dom';
import { cn } from '@/lib/utils';

interface ModalProps {
  isOpen: boolean;
  onClose: () => void;
  title: string;
  children: ReactNode;
  footer?: ReactNode;
  size?: 'sm' | 'md' | 'lg' | 'xl';
}

export function Modal({ isOpen, onClose, title, children, footer, size = 'md' }: ModalProps) {
  useEffect(() => {
    if (isOpen) {
      document.body.style.overflow = 'hidden';
      
      const handleEscape = (e: KeyboardEvent) => {
        if (e.key === 'Escape') onClose();
      };
      
      document.addEventListener('keydown', handleEscape);
      return () => {
        document.body.style.overflow = 'unset';
        document.removeEventListener('keydown', handleEscape);
      };
    }
  }, [isOpen, onClose]);

  if (!isOpen) return null;

  const sizeStyles = {
    sm: 'max-w-md',
    md: 'max-w-2xl',
    lg: 'max-w-4xl',
    xl: 'max-w-6xl'
  };

  return createPortal(
    <div className="fixed inset-0 z-50 overflow-y-auto">
      <div 
        className="fixed inset-0 bg-black/50 transition-opacity"
        onClick={onClose}
      />
      
      <div className="flex min-h-full items-center justify-center p-4">
        <div className={cn(
          'relative w-full bg-white rounded-xl shadow-2xl',
          sizeStyles[size]
        )}>
          <div className="flex items-center justify-between px-6 py-4 border-b border-black/10">
            <h2 className="text-xl font-semibold text-gray-900">{title}</h2>
            <button
              onClick={onClose}
              className="text-gray-400 hover:text-gray-600 transition-colors"
            >
              <svg className="w-6 h-6" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
              </svg>
            </button>
          </div>
          
          <div className="px-6 py-4">
            {children}
          </div>
          
          {footer && (
            <div className="px-6 py-4 border-t border-black/10 flex justify-end gap-3">
              {footer}
            </div>
          )}
        </div>
      </div>
    </div>,
    document.body
  );
}
```

```typescript
  options: SelectOption[];
  placeholder?: string;
```

IMPLEMENTATION:
```typescript
'use client';

import { useState, useRef, ChangeEvent } from 'react';
import Papa from 'papaparse';
import { Modal } from './ui/Modal';
import { Button } from './ui/Button';
import { Select } from './ui/Select';
import type { Business, D7LeadRow } from '@/types';
import { toast } from 'sonner';

interface ImportModalProps {
  isOpen: boolean;
  onClose: () => void;
  onImport: (businesses: Partial<Business>[]) => Promise<void>;
}

interface ColumnMapping {
  businessName: string;
  email: string;
  phone: string;
  city: string;
  state: string;
  googleReviews: string;
  yelpReviews: string;
  facebookReviews: string;
  website: string;
}

export function ImportModal({ isOpen, onClose, onImport }: ImportModalProps) {
  const [file, setFile] = useState<File | null>(null);
  const [csvData, setCsvData] = useState<any[]>([]);
  const [headers, setHeaders] = useState<string[]>([]);
  const [mapping, setMapping] = useState<ColumnMapping>({
    businessName: 'Business Name',
    email: 'Email',
    phone: 'Phone',
    city: 'City',
    state: 'State',
    googleReviews: 'Google Reviews',
    yelpReviews: 'Yelp Reviews',
    facebookReviews: 'Facebook Reviews',
    website: 'Website'
  });
  const [importing, setImporting] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const handleFileChange = (e: ChangeEvent<HTMLInputElement>) => {
    const selectedFile = e.target.files?.[0];
    if (!selectedFile) return;

    setFile(selectedFile);

    Papa.parse(selectedFile, {
      header: true,
      complete: (results) => {
        setHeaders(results.meta.fields || []);
        setCsvData(results.data.slice(0, 3));
        
        const detectedHeaders = results.meta.fields || [];
        const newMapping = { ...mapping };
        
        if (detectedHeaders.includes('Business Name')) newMapping.businessName = 'Business Name';
        if (detectedHeaders.includes('Email')) newMapping.email = 'Email';
        if (detectedHeaders.includes('Phone')) newMapping.phone = 'Phone';
        if (detectedHeaders.includes('City')) newMapping.city = 'City';
        if (detectedHeaders.includes('State')) newMapping.state = 'State';
        if (detectedHeaders.includes('Google Reviews')) newMapping.googleReviews = 'Google Reviews';
        if (detectedHeaders.includes('Yelp Reviews')) newMapping.yelpReviews = 'Yelp Reviews';
        if (detectedHeaders.includes('Facebook Reviews')) newMapping.facebookReviews = 'Facebook Reviews';
        if (detectedHeaders.includes('Website')) newMapping.website = 'Website';
        
        setMapping(newMapping);
      },
      error: (error) => {
        toast.error(`CSV parse error: ${error.message}`);
      }
    });
  };

  const handleImport = async () => {
    if (!file) return;

    setImporting(true);

    Papa.parse(file, {
      header: true,
      complete: async (results) => {
        try {
          const businesses: Partial<Business>[] = results.data
            .filter((row: any) => row[mapping.businessName])
            .map((row: any) => ({
              business_name: row[mapping.businessName] || '',
              email: row[mapping.email] || null,
              phone: row[mapping.phone] || null,
              city: row[mapping.city] || null,
              state: row[mapping.state] || null,
              google_reviews_count: parseInt(row[mapping.googleReviews]) || 0,
              yelp_reviews_count: parseInt(row[mapping.yelpReviews]) || 0,
              facebook_reviews_count: parseInt(row[mapping.facebookReviews]) || 0,
              website_url: row[mapping.website] || null,
              pipeline_stage: 'lead_identified' as const,
              email_status: 'not_sent' as const
            }));

          await onImport(businesses);
          
          toast.success(`Imported ${businesses.length} businesses successfully`);
          resetModal();
        } catch (error: any) {
          toast.error(`Import failed: ${error.message}`);
        } finally {
          setImporting(false);
        }
      }
    });
  };

  const resetModal = () => {
    setFile(null);
    setCsvData([]);
    setHeaders([]);
    setImporting(false);
    if (fileInputRef.current) fileInputRef.current.value = '';
    onClose();
  };

  const columnOptions = headers.map(h => ({ value: h, label: h }));

  return (
    <Modal
      isOpen={isOpen}
      onClose={resetModal}
      title="Import Businesses from CSV"
      size="lg"
      footer={
        <>
          <Button variant="secondary" onClick={resetModal}>
            Cancel
          </Button>
          <Button 
            onClick={handleImport} 
            disabled={!file || importing}
            loading={importing}
          >
            Import {csvData.length > 0 && `(${csvData.length}+ businesses)`}
          </Button>
        </>
      }
    >
      <div className="space-y-6">
        <div>
          <label className="block text-sm font-medium text-gray-700 mb-2">
            Upload CSV File
          </label>
          <input
            ref={fileInputRef}
            type="file"
            accept=".csv"
            onChange={handleFileChange}
            className="block w-full text-sm text-gray-900 border border-black/10 rounded-xl cursor-pointer bg-white focus:outline-none p-2"
          />
          <p className="mt-1 text-xs text-gray-500">
            Supports D7 Lead Finder CSV exports
          </p>
        </div>

        {headers.length > 0 && (
          <div className="space-y-4">
            <h3 className="text-sm font-semibold text-gray-900">Map CSV Columns</h3>
            <div className="grid grid-cols-2 gap-4">
              <Select
                label="Business Name"
                value={mapping.businessName}
                onChange={(v) => setMapping({ ...mapping, businessName: v })}
                options={columnOptions}
              />
              <Select
                label="Email"
                value={mapping.email}
                onChange={(v) => setMapping({ ...mapping, email: v })}
                options={columnOptions}
              />
              <Select
                label="Phone"
                value={mapping.phone}
                onChange={(v) => setMapping({ ...mapping, phone: v })}
                options={columnOptions}
              />
              <Select
                label="City"
                value={mapping.city}
                onChange={(v) => setMapping({ ...mapping, city: v })}
                options={columnOptions}
              />
              <Select
                label="State"
                value={mapping.state}
                onChange={(v) => setMapping({ ...mapping, state: v })}
                options={columnOptions}
              />
              <Select
                label="Google Reviews"
                value={mapping.googleReviews}
                onChange={(v) => setMapping({ ...mapping, googleReviews: v })}
                options={columnOptions}
              />
              <Select
                label="Yelp Reviews"
                value={mapping.yelpReviews}
                onChange={(v) => setMapping({ ...mapping, yelpReviews: v })}
                options={columnOptions}
              />
              <Select
                label="Facebook Reviews"
                value={mapping.facebookReviews}
                onChange={(v) => setMapping({ ...mapping, facebookReviews: v })}
                options={columnOptions}
              />
            </div>
          </div>
        )}

        {csvData.length > 0 && (
          <div>
            <h3 className="text-sm font-semibold text-gray-900 mb-2">Preview (first 3 rows)</h3>
            <div className="overflow-x-auto border border-black/10 rounded-xl">
              <table className="min-w-full divide-y divide-black/10">
                <thead className="bg-gray-50">
                  <tr>
                    {Object.keys(csvData[0]).slice(0, 5).map(key => (
                      <th key={key} className="px-4 py-2 text-left text-xs font-medium text-gray-700 uppercase">
                        {key}
                      </th>
                    ))}
                  </tr>
                </thead>
                <tbody className="bg-white divide-y divide-black/10">
                  {csvData.map((row, idx) => (
                    <tr key={idx}>
                      {Object.values(row).slice(0, 5).map((val: any, i) => (
                        <td key={i} className="px-4 py-2 text-sm text-gray-900">
                          {val}
                        </td>
                      ))}
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}
      </div>
    </Modal>
  );
}
```

```typescript
'use client';

import { useState, useEffect } from 'react';
import { Input } from './ui/Input';
import { Button } from './ui/Button';
import type { FilterState, PipelineStage, EmailStatus, PricingTier } from '@/types';
import { PIPELINE_STAGES } from '@/types';

interface FilterBarProps {
  filters: FilterState;
  onFiltersChange: (filters: FilterState) => void;
  onClear: () => void;
}

export function FilterBar({ filters, onFiltersChange, onClear }: FilterBarProps) {
  const [searchInput, setSearchInput] = useState(filters.search);

  useEffect(() => {
    const timer = setTimeout(() => {
      onFiltersChange({ ...filters, search: searchInput });
    }, 300);

    return () => clearTimeout(timer);
  }, [searchInput]);

  const hasActiveFilters = 
    filters.search || 
    filters.stages.length > 0 || 
    filters.emailStatuses.length > 0 || 
    filters.tags.length > 0 ||
    filters.pricingTiers.length > 0 ||
    filters.cities.length > 0;

  const toggleStage = (stage: PipelineStage) => {
    const newStages = filters.stages.includes(stage)
      ? filters.stages.filter(s => s !== stage)
      : [...filters.stages, stage];
    onFiltersChange({ ...filters, stages: newStages });
  };

  const toggleEmailStatus = (status: EmailStatus) => {
    const newStatuses = filters.emailStatuses.includes(status)
      ? filters.emailStatuses.filter(s => s !== status)
      : [...filters.emailStatuses, status];
    onFiltersChange({ ...filters, emailStatuses: newStatuses });
  };

  const togglePricingTier = (tier: PricingTier) => {
    const newTiers = filters.pricingTiers.includes(tier)
      ? filters.pricingTiers.filter(t => t !== tier)
      : [...filters.pricingTiers, tier];
    onFiltersChange({ ...filters, pricingTiers: newTiers });
  };

  return (
    <div className="bg-white rounded-xl border border-black/10 p-4">
      <div className="flex flex-wrap items-center gap-3">
        <div className="w-64">
          <Input
            placeholder="Search businesses..."
            value={searchInput}
            onChange={(e) => setSearchInput(e.target.value)}
            icon={
              <svg className="w-4 h-4" fill="none" viewBox="0 0 24 24" stroke="currentColor">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M21 21l-6-6m2-5a7 7 0 11-14 0 7 7 0 0114 0z" />
              </svg>
            }
          />
        </div>

        <details className="relative">
          <summary className="cursor-pointer px-4 py-2 rounded-xl border border-black/10 bg-white hover:bg-black/5 transition-colors list-none flex items-center gap-2">
            <span className="text-sm font-medium">Pipeline Stage</span>
            {filters.stages.length > 0 && (
              <span className="px-2 py-0.5 rounded-full bg-blue-100 text-blue-700 text-xs font-medium">
                {filters.stages.length}
              </span>
            )}
            <svg className="w-4 h-4 ml-1" fill="none" viewBox="0 0 24 24" stroke="currentColor">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" />
            </svg>
          </summary>
          <div className="absolute z-10 mt-2 w-64 bg-white rounded-xl shadow-lg border border-black/10 p-2 max-h-96 overflow-y-auto">
            {PIPELINE_STAGES.map(stage => (
              <label key={stage.stage} className="flex items-center gap-2 px-3 py-2 hover:bg-gray-50 rounded-lg cursor-pointer">
                <input
                  type="checkbox"
                  checked={filters.stages.includes(stage.stage)}
                  onChange={() => toggleStage(stage.stage)}
                  className="rounded border-gray-300 text-blue-600 focus:ring-blue-500"
                />
                <span className="text-sm text-gray-900">{stage.label}</span>
              </label>
            ))}
          </div>
        </details>

        <details className="relative">
          <summary className="cursor-pointer px-4 py-2 rounded-xl border border-black/10 bg-white hover:bg-black/5 transition-colors list-none flex items-center gap-2">
            <span className="text-sm font-medium">Email Status</span>
            {filters.emailStatuses.length > 0 && (
              <span className="px-2 py-0.5 rounded-full bg-blue-100 text-blue-700 text-xs font-medium">
                {filters.emailStatuses.length}
              </span>
            )}
            <svg className="w-4 h-4 ml-1" fill="none" viewBox="0 0 24 24" stroke="currentColor">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" />
            </svg>
          </summary>
          <div className="absolute z-10 mt-2 w-48 bg-white rounded-xl shadow-lg border border-black/10 p-2">
            {(['not_sent', 'sent', 'opened', 'clicked', 'replied', 'bounced', 'unsubscribed'] as EmailStatus[]).map(status => (
              <label key={status} className="flex items-center gap-2 px-3 py-2 hover:bg-gray-50 rounded-lg cursor-pointer">
                <input
                  type="checkbox"
                  checked={filters.emailStatuses.includes(status)}
                  onChange={() => toggleEmailStatus(status)}
                  className="rounded border-gray-300 text-blue-600 focus:ring-blue-500"
                />
                <span className="text-sm text-gray-900 capitalize">{status.replace('_', ' ')}</span>
              </label>
            ))}
          </div>
        </details>

        <details className="relative">
          <summary className="cursor-pointer px-4 py-2 rounded-xl border border-black/10 bg-white hover:bg-black/5 transition-colors list-none flex items-center gap-2">
            <span className="text-sm font-medium">Pricing Tier</span>
            {filters.pricingTiers.length > 0 && (
              <span className="px-2 py-0.5 rounded-full bg-blue-100 text-blue-700 text-xs font-medium">
                {filters.pricingTiers.length}
              </span>
            )}
            <svg className="w-4 h-4 ml-1" fill="none" viewBox="0 0 24 24" stroke="currentColor">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" />
            </svg>
          </summary>
          <div className="absolute z-10 mt-2 w-48 bg-white rounded-xl shadow-lg border border-black/10 p-2">
            {(['bronze', 'silver', 'gold'] as PricingTier[]).map(tier => (
              <label key={tier} className="flex items-center gap-2 px-3 py-2 hover:bg-gray-50 rounded-lg cursor-pointer">
                <input
                  type="checkbox"
                  checked={filters.pricingTiers.includes(tier)}
                  onChange={() => togglePricingTier(tier)}
                  className="rounded border-gray-300 text-blue-600 focus:ring-blue-500"
                />
                <span className="text-sm text-gray-900 capitalize">
                  {tier} (${tier === 'bronze' ? '125' : tier === 'silver' ? '110' : '90'}/review)
                </span>
              </label>
            ))}
          </div>
        </details>

        {hasActiveFilters && (
          <Button variant="ghost" size="sm" onClick={onClear}>
            Clear Filters
          </Button>
        )}
      </div>
    </div>
  );
}
```
